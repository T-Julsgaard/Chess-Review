import {readFile} from 'node:fs/promises';
import {gunzipSync} from 'node:zlib';
import assert from 'node:assert/strict';
import {Chess} from '../../../../lib/chess.js';
import {bindings} from '../../E138-bounded-engine-panels/code/source-bindings.mjs';
export const dir='research/experiments/E166-development-defender-capacity',seeds=[dir+'/code/development-panel.mjs','research/experiments/E153-second-target-defense/code/panel.mjs'];
export const key=i=>i.attackContextMode==='development'?JSON.stringify(['development',i.fen,i.history,i.move,i.developmentAlternativeHistory,i.developmentAttackPlies??2]):JSON.stringify(['defender',i.fen,i.history,[i.move,i.capacityAlternative].sort(),i.capacityTargets]);
const qkey=(input,H)=>JSON.stringify([input.fen,input.history,input.move,H]);
export async function savedPanels(){const hashes=await bindings(seeds),cache=new Map(),queries=new Map(),store=(input,raw)=>{cache.set(key(input),{input,raw});if(input.attackContextMode==='development')for(const v of raw.variants)queries.set(qkey(v.input,raw.plies),v);};const legacy=JSON.parse(gunzipSync(await readFile('research/experiments/E153-second-target-defense/evidence/observations.json.gz'))),oldHashes=await bindings(['research/experiments/E153-second-target-defense/code/panel.mjs']);for(const [p,h]of Object.entries(oldHashes))assert.equal(legacy.sourceHashes[p],h,'Changed E153 collection source');for(const r of legacy.rows){const k=r.key;cache.set(JSON.stringify(['defender',k.fen,k.history,k.moves,k.targets]),{raw:r.panel});}for(const name of ['smoke','observations']){let p;try{p=JSON.parse(gunzipSync(await readFile(dir+'/evidence/'+name+'.json.gz')));}catch(e){if(e.code==='ENOENT')continue;throw e;}for(const [name,h]of Object.entries(hashes))assert.equal(p.sourceHashes[name],h,'Changed E166 collector source');for(const r of p.rows){const raw=r.raw||r.result?.attackContextAnalysis.witness?.raw;if(raw)store(r.input,raw);}}const get=input=>{const found=cache.get(key(input));if(found)return found;if(input.attackContextMode!=='development'||!input.developmentAlternativeHistory)return null;const ah=input.developmentAlternativeHistory,c=new Chess(ah.fen);for(const m of ah.moves)c.move(m);const H=input.developmentAttackPlies??2,variants=[{fen:input.fen,history:input.history,move:input.move},{fen:c.fen(),history:ah,move:input.move}].map(f=>queries.get(qkey(f,H)));if(variants.some(v=>!v))return null;return{raw:{schema:'E166-paired-development-attack-v1',plies:H,variants,nodes:2+variants.reduce((n,v)=>n+2+v.input.history.moves.length+v.query.searchTrace.length,0)}};};return{hashes,cache,get,store};}
