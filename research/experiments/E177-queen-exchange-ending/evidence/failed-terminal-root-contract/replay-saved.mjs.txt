import assert from 'node:assert/strict';import {readFile} from 'node:fs/promises';import {gunzipSync} from 'node:zlib';import {openResearchData,sha256} from '../../../data-policy.mjs';import {bindings} from '../../E138-bounded-engine-panels/code/source-bindings.mjs';import {checkWitness} from './check-witness.mjs';import {fixtures,prefixControl} from './fixtures.mjs';
// Independent reconstruction; no detector, collector, context or derive imports.
const data=await openResearchData(['D001'],{purpose:'test'}),dir='research/experiments/E177-queen-exchange-ending',run=JSON.parse(await readFile(dir+'/evidence/run.json','utf8')),build=JSON.parse(await readFile(dir+'/build.json','utf8')),outputs={};assert.deepEqual(run.datasetReceipt,data.receipt);assert.deepEqual(run.sourceHashes,build.inputHashes);assert.deepEqual(await bindings(Object.keys(run.sourceHashes)),run.sourceHashes);
for(const name of ['observations','results']){const bytes=await readFile(dir+'/evidence/'+name+'.json.gz');assert.equal(sha256(bytes),run.outputs[name]);outputs[name]=JSON.parse(gunzipSync(bytes));}
const raw=outputs.observations;assert.deepEqual(raw.datasetReceipt,data.receipt);assert.deepEqual(await bindings(Object.keys(raw.sourceHashes)),raw.sourceHashes);
const old=JSON.parse(gunzipSync(await readFile(raw.queryReuseAudit.sourceObservation)));assert.equal(sha256(await readFile(raw.queryReuseAudit.sourceObservation)),raw.queryReuseAudit.sourceObservationSha256);assert.equal(old.rows.length,raw.queryReuseAudit.checked);const matches=old.rows.filter(r=>fixtures.some(i=>r.key.fen===i.fen&&JSON.stringify(r.key.history)===JSON.stringify(i.history)&&r.key.plies===(i.endingMatePlies??2)&&[i.move,i.endingAlternative].includes(r.key.move)));assert.equal(matches.length,0);assert.equal(raw.queryReuseAudit.matched,0);
const key=i=>JSON.stringify([i.fen,i.history,i.move,i.endingAlternative,i.endingMatePlies===undefined?2:i.endingMatePlies]),cache=new Map(raw.rows.map(r=>[key(r.input),r.panel])),seen=new Set();let witnesses=0,positives=0;
assert.equal(outputs.results.rows.length,fixtures.length);
for(const [index,{input,result,inherited}]of outputs.results.rows.entries()){
  assert.deepEqual(input,JSON.parse(JSON.stringify(fixtures[index])));assert.deepEqual(result.events.filter(e=>e.evidence?.experiment==='E177').map(e=>e.id),input.expected);assert.deepEqual(result.events.filter(e=>e.evidence?.experiment!=='E177'),inherited.events);assert.equal(result.before,inherited.before);assert.equal(result.after,inherited.after);if(input.expected.length)positives++;
  const a=result.endingPreparationAnalysis;assert.equal(a.limit,input.maxEndingPreparationNodes??50000);assert.ok(a.nodes>=0&&a.nodes<=a.limit+1);
  if(a.witness){assert.deepEqual(cache.get(key(input)),a.witness.panel);checkWitness(input,result);seen.add(key(input));witnesses++;}
  else{assert.equal(a.witness,null);assert.equal(result.comment,inherited.comment);assert.equal(input.expected.length,0);const status=input.maxEndingPreparationNodes===0?'exhausted':input.history===undefined?'history-prerequisite':input.endingAlternative===undefined?'alternative-prerequisite':input.endingMatePlies===0?'horizon-prerequisite':'material-prerequisite';assert.equal(a.status,status);}
}
const smoke=JSON.parse(gunzipSync(await readFile(dir+'/evidence/smoke.json.gz')));assert.deepEqual(smoke.datasetReceipt,data.receipt);assert.deepEqual(smoke.sourceHashes,raw.sourceHashes);assert.deepEqual(smoke.queryReuseAudit,raw.queryReuseAudit);for(const row of smoke.rows){assert.equal(row.error,undefined);const same=outputs.results.rows.find(r=>r.input.id===row.input.id);assert.deepEqual(row.input,same.input);assert.deepEqual(row.result,same.result);checkWitness(row.input,row.result);}
assert.equal(witnesses,run.witnesses);assert.equal(seen.size,raw.rows.length);assert.equal(positives,run.positives);assert.equal(run.cases,fixtures.length);assert.equal(run.reused+run.fresh,witnesses);assert.equal(run.preregistration,'81eab09');assert.match(run.revision,/^[a-f0-9]{40}$/);assert.match(run.argv.at(-1),/pilot\.mjs$/);assert.equal(run.engine,null);assert.equal(run.seed,null);
console.log(JSON.stringify({cases:run.cases,witnesses,positives,sourceHashes:Object.keys(run.sourceHashes).length,independentReplay:'passed'}));
const prefix=JSON.parse(gunzipSync(await readFile(dir+'/evidence/prefix-control.json.gz')));assert.deepEqual(prefix.datasetReceipt,data.receipt);assert.deepEqual(prefix.sourceHashes,raw.sourceHashes);assert.deepEqual(prefix.input,prefixControl);checkWitness(prefix.input,prefix.result);console.log(JSON.stringify({extraPrefixWitness:'passed'}));
